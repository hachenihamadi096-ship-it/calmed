// ════════════════════════════════════════════════════════════
//  SETUP  (runs on Core 1)
// ════════════════════════════════════════════════════════════
void setup() {
    Serial.begin(115200);
    delay(500);   // give Serial time to connect before any crash
    Serial.println("\n[CAL-MED] " FW_VERSION " booting...");
    Serial.printf("[MEM] Free heap: %u  PSRAM: %u\n",
                  ESP.getFreeHeap(), ESP.getFreePsram());
    dataMutex = xSemaphoreCreateMutex();
    spiMutex  = xSemaphoreCreateMutex();
    configASSERT(dataMutex);
    configASSERT(spiMutex);

    sensorsInit();
    loadLimits();   // restore user-configured alert thresholds from NVS
    displayInit();
    touchInit();

    drawBoot();
    screenStart   = millis();
    lastActivity  = millis();
    energyStartMs = millis();

    xTaskCreatePinnedToCore(
        sensorTask, "SensorTask",
        SENSOR_STACK, nullptr,
        SENSOR_PRIORITY, nullptr,
        SENSOR_TASK_CORE
    );

    xTaskCreatePinnedToCore(
        wifiTask, "WiFiTask",
        WIFI_STACK, nullptr,
        WIFI_PRIORITY, nullptr,
        WIFI_TASK_CORE
    );

    xTaskCreatePinnedToCore(
        sdLogTask, "SDTask",
        SD_STACK, nullptr,
        SD_PRIORITY, nullptr,
        1  // Core 1 — keep off Core 0 (SensorTask)
    );

    mqttInit();
    xTaskCreatePinnedToCore(
        mqttTask, "MQTTTask",
        MQTT_STACK, nullptr,
        MQTT_PRIORITY, nullptr,
        MQTT_TASK_CORE
    );

    xTaskCreatePinnedToCore(
        displayTask, "DisplayTask",
        DISPLAY_STACK, nullptr,
        DISPLAY_PRIORITY, nullptr,
        DISPLAY_TASK_CORE
    );

    Serial.println("[OK] Tasks started");
}

void loop() { vTaskDelay(pdMS_TO_TICKS(1000)); }

// ════════════════════════════════════════════════════════════
//  DISPLAY / UI TASK  (Core 1)
// ════════════════════════════════════════════════════════════
void displayTask(void* param) {
    for (;;) {
        processTouch();

        // Idle timeout → back to menu
        if (currentScreen != SCR_BOOT        &&
            currentScreen != SCR_VOLT_SELECT &&
            currentScreen != SCR_MENU) {
            if (millis() - lastActivity >= SCREEN_TIMEOUT_MS) {
                resetDashboard(); resetPower();
                resetEnergy();    resetAlerts();
                resetSettings();
                gotoScreen(SCR_MENU);
            }
        }

        switch (currentScreen) {

            case SCR_BOOT:
                updateBoot(millis() - screenStart);
                break;

            case SCR_VOLT_SELECT:
                if (needRedraw) { drawVoltSelect();   needRedraw = false; }
                if (tapped)       handleVoltSelect(touchX, touchY);
                break;

            case SCR_SENSOR_SELECT:
                if (needRedraw) { drawSensorSelect();  needRedraw = false; }
                if (tapped)       handleSensorSelect(touchX, touchY);
                break;

            case SCR_MENU:
                if (needRedraw) { drawMenu(); needRedraw = false; }
                if (tapped)     { handleMenu(touchX, touchY); }
                else            { refreshMenuDot(); }
                break;

            case SCR_DASHBOARD:
                drawDashboard();
                if (tapped) handleDashboard(touchX, touchY);
                break;

            case SCR_POWER:
                drawPower();
                if (tapped) handlePower(touchX, touchY);
                break;

            case SCR_ENERGY:
                drawEnergy();
                if (tapped) handleEnergy(touchX, touchY);
                break;

            case SCR_ALERTS:
                drawAlerts();
                if (tapped) handleAlerts(touchX, touchY);
                break;

            case SCR_SETTINGS:
                if (needRedraw) { drawSettings(); needRedraw = false; }
                if (tapped)       handleSettings(touchX, touchY);
                else {
                    static unsigned long _sLast = 0;
                    if (millis() - _sLast >= 2000) {
                        _sLast = millis();
                        drawSettings();
                    }
                }
                break;

            case SCR_WIFI_SETUP:
                drawWifi();
                if (tapped) handleWifi(touchX, touchY);
                break;

            case SCR_LIMITS_EDIT:
                drawLimits();
                if (tapped) handleLimits(touchX, touchY);
                break;

            default:
                gotoScreen(SCR_MENU);
                break;
        }

        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

uint16_t threshColor(float val, float nominal, float toleranceFrac) {
    float dev = fabsf(val - nominal) / nominal;
    if (dev < toleranceFrac * 0.5f) return C_GREEN;
    if (dev < toleranceFrac)        return C_YELLOW;
    return C_RED;
}

uint16_t pfColor(float pf) {
    if (pf >= lim_pf_warn) return C_GREEN;
    if (pf >= lim_pf_crit) return C_YELLOW;
    return C_RED;
}

uint16_t unbalColor(float pct) {

// ── Simultaneous V+I → P, Q, S, PF, angle ────────────────
void calcPower(int idx) {
    static int16_t vBuf[NUM_SAMPLES];
    static int16_t iBuf[NUM_SAMPLES];

    uint32_t next = micros();
    for (int i = 0; i < NUM_SAMPLES; i++) {
        while ((int32_t)(micros() - next) < 0);
        next += INTERVAL_US;
        vBuf[i] = (int16_t)analogRead(vPins[idx]);
        iBuf[i] = (int16_t)analogRead(iPins[idx]);
        if (i % 50 == 49) taskYIELD();   // yield every 50 samples (~2.5ms)
    }

    int32_t vSum = 0, iSum = 0;
    for (int i = 0; i < NUM_SAMPLES; i++) { vSum += vBuf[i]; iSum += iBuf[i]; }
    float vDC = (float)vSum / NUM_SAMPLES;
    float iDC = (float)iSum / NUM_SAMPLES;

    float vSS = 0, iSS = 0, viSum = 0;
    for (int i = 0; i < NUM_SAMPLES; i++) {
        float cv = (voltMode == 2) ? calV400[idx] : calV230[idx];
        float v = ((float)vBuf[i] - vDC) * cv;
        float c = ((float)iBuf[i] - iDC) * calIph[idx];
        vSS   += v * v;
        iSS   += c * c;
        viSum += v * c;
    }

    float vrms = sqrtf(vSS / NUM_SAMPLES);
    float irms = sqrtf(iSS / NUM_SAMPLES);
    float P    = viSum / NUM_SAMPLES;
    float S    = vrms * irms;
    float Q    = sqrtf(fabsf(S * S - P * P));
    float pf   = (S > 0.01f) ? (P / S) : 1.0f;
    pf = constrain(pf, -1.0f, 1.0f);

    if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(5)) == pdTRUE) {
        phase[idx].Vrms  = vrms;
        phase[idx].Irms  = irms;
        phase[idx].P     = P;
        phase[idx].Q     = Q;
        phase[idx].S     = S;
        phase[idx].PF    = pf;
        phase[idx].angle = acosf(fabsf(pf));
        xSemaphoreGive(dataMutex);
    }
}

// ── Frequency (zero-cross counting, 200 ms window) ────────
float measureFreq(int pin) {
    const int midADC = 2048;
    bool lastHigh   = (analogRead(pin) > midADC);
    uint32_t t0     = millis();
    int crosses     = 0;
    uint32_t first  = 0, last = 0;

    while (millis() - t0 < 200) {
        bool isHigh = (analogRead(pin) > midADC);
        if (!lastHigh && isHigh) {
            if (crosses == 0) first = micros();
            last = micros();
            crosses++;
        }
        lastHigh = isHigh;
        vTaskDelay(pdMS_TO_TICKS(1));
    }
    if (crosses < 2) return 50.0f;
    float period = (float)(last - first) / (float)(crosses - 1);
    return 1e6f / period;
}

// ── IEC unbalance % ───────────────────────────────────────
float calcUnbalance(float a, float b, float c) {
    float avg = (a + b + c) / 3.0f;
    if (avg < 1.0f) return 0.0f;
    float maxDev = max({fabsf(a - avg), fabsf(b - avg), fabsf(c - avg)});
    return (maxDev / avg) * 100.0f;
}

// ── Alert evaluation ──────────────────────────────────────
void updateAlerts() {
    float vnom = (voltMode == 2) ? V_NOMINAL_400 : V_NOMINAL_230;
    alertVunbal  = (vUnbal >= lim_vunbal_warn);
    alertPF      = (avgPF > 0.01f && avgPF < lim_pf_warn);
    alertOverV   = false;
    alertUnderV  = false;
    alertOverC   = false;

    float iRated = (sensorMode >= 0 && sensorMode < 6)
                   ? SCT_RATED_A[sensorMode] : 30.0f;

    for (int i = 0; i < 3; i++) {
        if (phase[i].Vrms > vnom * (1.0f + lim_ov_pct / 100.0f))
            alertOverV = true;
        if (phase[i].Vrms < vnom * (1.0f - lim_uv_pct / 100.0f) && phase[i].Vrms > 5.0f)
            alertUnderV = true;
        if (phase[i].Irms > iRated * (lim_oc_warn / 100.0f))
            alertOverC = true;
    }
    activeAlertCount = (int)alertVunbal + (int)alertPF
                     + (int)alertOverV  + (int)alertUnderV
                     + (int)alertOverC;
}

// ════════════════════════════════════════════════════════════
//  FREERTOS SENSOR TASK  (Core 0)
//  Cycles through phases, updates all aggregates, yields.
// ════════════════════════════════════════════════════════════
void sensorTask(void* param) {
    unsigned long lastEnergy = millis();

    for (;;) {
        int phases = (voltMode == 0) ? 1 : 3;

        // ── Real ADC ────────────────────────────────────
        for (int i = 0; i < phases; i++) calcPower(i);

        float f = measureFreq(vPins[0]);

        float tp = 0, ts2 = 0, tq = 0;
        for (int i = 0; i < phases; i++) {
            if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(5)) == pdTRUE) {
                tp  += phase[i].P;
                ts2 += phase[i].S;
                tq  += phase[i].Q;
                xSemaphoreGive(dataMutex);
            }
        }
        float apf = (ts2 > 0.01f) ? (tp / ts2) : 1.0f;
        float vu  = 0, iu = 0;
        if (phases == 3) {
            float va = phase[0].Vrms, vb = phase[1].Vrms, vc = phase[2].Vrms;
            float ia = phase[0].Irms, ib = phase[1].Irms, ic = phase[2].Irms;
            vu = calcUnbalance(va, vb, vc);
            iu = calcUnbalance(ia, ib, ic);
        }

        unsigned long now = millis();
        float dt_h = (float)(now - lastEnergy) / 3600000.0f;
        lastEnergy = now;

        if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(10)) == pdTRUE) {
            freqHz = f;
            totalP = tp; totalS = ts2; totalQ = tq;
            avgPF  = apf;
            vUnbal = vu; iUnbal = iu;
            for (int i = 0; i < phases; i++)
                energyWh[i] += phase[i].P * dt_h;
            totalEnergyWh = energyWh[0] + energyWh[1] + energyWh[2];
            xSemaphoreGive(dataMutex);
        }

        updateAlerts();
        vTaskDelay(pdMS_TO_TICKS(50));   // ~20 Hz update rate, yields Core 0
    }
}

//  Tap fires on PRESS (not release) after TOUCH_DEBOUNCE_MS.

static bool          _down        = false;
static unsigned long _lastTapMs   = 0;

    // ts is initialised in displayInit() — nothing extra needed
}

void processTouch() {
    tapped = false;

    bool isDown = false;
    if (ts.tirqTouched() && ts.touched()) {
        TS_Point p = ts.getPoint();
        if (p.z >= 100) {
            // Rotation 2 (180°): raw axes are fully inverted on both axes
            int mx = map(p.x, TOUCH_X_MAX, TOUCH_X_MIN, 0, TFT_W - 1);
            int my = map(p.y, TOUCH_Y_MAX, TOUCH_Y_MIN, 0, TFT_H - 1);
            touchX = constrain(mx, 0, TFT_W - 1);
            touchY = constrain(my, 0, TFT_H - 1);
            isDown = true;
        }
    }

    _prevDown = _down;
    _down     = isDown;

    // Tap = rising edge (press) after debounce
    if (!_prevDown && _down) {
        unsigned long now = millis();
        if (now - _lastTapMs >= TOUCH_DEBOUNCE_MS) {
            _lastTapMs   = now;
            tapped       = true;
            lastActivity = now;   // reset idle timeout
        }
    }
}

    Serial.printf("[MQTT] Broker set to %s:%d\n", MQTT_BROKER_IP, MQTT_BROKER_PORT);

// ── FreeRTOS task ────────────────────────────────────────────────
void mqttTask(void* param) {
    Serial.println("[MQTT] Task started — waiting for WiFi...");
    while (WiFi.status() != WL_CONNECTED) {
        vTaskDelay(pdMS_TO_TICKS(500));
    }
    Serial.println("[MQTT] WiFi ready — starting MQTT loop.");

    unsigned long lastPublish   = 0;
    unsigned long lastReconnect = 0;

    for (;;) {
        // ── Pause during WiFi network switch ─────────────────
        if (wifiSwitching) {
            _mqttConnected = false;
            _mqtt.disconnect();        // cleanly close the socket NOW
            _wifiClient.stop();
            vTaskDelay(pdMS_TO_TICKS(200));
            continue;
        }

        bool wifiUp = (WiFi.status() == WL_CONNECTED);

        if (!wifiUp) {
            _mqttConnected = false;
            vTaskDelay(pdMS_TO_TICKS(500));
            continue;
        }

        // ── Reconnect if needed (retry every 15s) ────────────────
        if (!_mqtt.connected()) {
            _mqttConnected = false;
            unsigned long now = millis();
            if (now - lastReconnect >= 15000) {
                lastReconnect = now;
                _wifiClient.stop();
                vTaskDelay(pdMS_TO_TICKS(100));  // yield before blocking connect
                _mqttConnected = mqttReconnect();

                // Trigger SD replay once after reconnect
                if (_mqttConnected && !_replayDone) {
                    _replayDone = true;
                    sdReplayAndClear();
                }
            }
        } else {
            _mqttConnected = true;
            _mqtt.loop();

            // ── Publish live data every 1000ms ───────────────────
            if (millis() - lastPublish >= 1000) {
                lastPublish = millis();
                publishData();
                publishAlerts();
            }
        }

        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

//  sd_logger.cpp  —  SD card store-and-forward
//  Uses FSPI (SPI2) — ESP32-S3 has no VSPI, only FSPI and HSPI
//  SD wiring: SCK=39, MOSI=40, MISO=41, CS=42
//
//  Store-and-forward behaviour:
//    WiFi/MQTT UP   → nothing written to SD
//    WiFi/MQTT DOWN → buffers rows to /calmed/buffer.csv every 1s
//    WiFi/MQTT back → mqttTask calls sdReplayAndClear():
//                       publishes buffered rows to calmed/data/replay,
//                       then deletes buffer file
// ╚══════════════════════════════════════════════════════════════════╝
#include "sd_logger.h"
#include "globals.h"
#include "config.h"
#include <SPI.h>
#include <SD.h>
#include <WiFi.h>

// ESP32-S3: VSPI doesn't exist — use FSPI (SPI2)
#ifndef FSPI
  #define FSPI 0
#endif

#define SD_SCK   38
#define SD_MOSI  41
#define SD_MISO  47
#define SD_CS    42

#define LOG_DIR      "/calmed"
#define LOG_INTERVAL 1000
#define FLUSH_EVERY  5
#define BUFFER_FILE  "/calmed/buffer.csv"

static SPIClass sdSPI(FSPI);
static bool    _sdReady   = false;
static bool    _buffering = false;
static File    _bufFile;
static int     _bufFlush  = 0;

bool sdReady()      { return _sdReady; }

void sdInit() {
    sdSPI.begin(SD_SCK, SD_MISO, SD_MOSI, SD_CS);
    if (!SD.begin(SD_CS, sdSPI, 400000)) {
        Serial.println("[SD] No card or init failed — buffering disabled");
        _sdReady = false;
        return;
    }
    Serial.printf("[SD] Card ready — %llu MB\n", SD.cardSize() / (1024*1024));
    if (!SD.exists(LOG_DIR)) SD.mkdir(LOG_DIR);
    _sdReady = true;
}

// ── Helper: snapshot current data into a CSV row string ──────────
static void buildRow(char* row, size_t sz) {
    PhaseData snap[3];
    float tp, ts2, tq, vu, f;
    bool avun, apf, aov, auv;
    if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(20)) != pdTRUE) {
        row[0] = '\0'; return;
    }
    for (int i = 0; i < 3; i++) snap[i] = phase[i];
    tp = totalP; ts2 = totalS; tq = totalQ; vu = vUnbal; f = freqHz;
    avun = alertVunbal; apf = alertPF;
    aov  = alertOverV;  auv = alertUnderV;
    xSemaphoreGive(dataMutex);

    char alerts[32] = "none";
    if (avun || apf || aov || auv)
        snprintf(alerts, sizeof(alerts), "%s%s%s%s",
            avun ? "UNBAL " : "", apf ? "LOWPF " : "",
            aov  ? "OVERV " : "", auv ? "UNDERV" : "");

    snprintf(row, sz,
        "%lu,%.2f,%.2f,%.2f,%.3f,%.3f,%.3f,"
        "%.1f,%.1f,%.1f,%.1f,%.1f,%.1f,"
        "%.3f,%.3f,%.3f,%.1f,%.1f,%.1f,%.2f,%.2f,%s",
        millis(),
        snap[0].Vrms, snap[1].Vrms, snap[2].Vrms,
        snap[0].Irms, snap[1].Irms, snap[2].Irms,
        snap[0].P,    snap[1].P,    snap[2].P,
        snap[0].S,    snap[1].S,    snap[2].S,
        snap[0].PF,   snap[1].PF,   snap[2].PF,
        tp, ts2, tq, vu, f, alerts);
}

// ── Replay buffer → MQTT then delete ─────────────────────────────
void sdReplayAndClear() {
    if (!_sdReady) return;
    if (!SD.exists(BUFFER_FILE)) return;
    if (_bufFile) { _bufFile.flush(); _bufFile.close(); }
    _buffering = false;

    File f = SD.open(BUFFER_FILE, FILE_READ);
    if (!f) { Serial.println("[SD] Replay: cannot open buffer"); return; }

    Serial.println("[SD] Replaying buffered data to MQTT...");
    int count = 0;

    extern void mqttPublishRaw(const char* topic, const char* payload);

    while (f.available()) {
        String line = f.readStringUntil('\n');
        line.trim();
        if (line.length() < 10) continue;
        char payload[280];
        snprintf(payload, sizeof(payload),
            "{\"buffered\":true,\"csv\":\"%s\"}", line.c_str());
        mqttPublishRaw("calmed/data/replay", payload);
        count++;
        vTaskDelay(pdMS_TO_TICKS(50));
    }
    f.close();
    SD.remove(BUFFER_FILE);
    Serial.printf("[SD] Replay done — %d rows sent, buffer cleared\n", count);
}

// ── Attempt to recover a failed SD card ──────────────────────────
static void sdRecover() {
    Serial.println("[SD] Card error — attempting remount...");
    _sdReady = false;
    if (_bufFile) { _bufFile.close(); }
    _buffering = false;
    SD.end();
    vTaskDelay(pdMS_TO_TICKS(1000));
    sdSPI.begin(SD_SCK, SD_MISO, SD_MOSI, SD_CS);
    if (SD.begin(SD_CS, sdSPI, 400000)) {
        if (!SD.exists(LOG_DIR)) SD.mkdir(LOG_DIR);
        _sdReady = true;
        Serial.println("[SD] Remount OK");
    } else {
        Serial.println("[SD] Remount failed — card absent or corrupt");
    }
}

// ── FreeRTOS task ─────────────────────────────────────────────────
void sdLogTask(void* param) {
    vTaskDelay(pdMS_TO_TICKS(5000));   // wait for boot + sensors to settle
    sdInit();

    unsigned long lastLog = 0;

    for (;;) {
        if (!_sdReady) {
            vTaskDelay(pdMS_TO_TICKS(5000));
            sdRecover();
            continue;
        }
        if (millis() - lastLog < LOG_INTERVAL) { vTaskDelay(pdMS_TO_TICKS(100)); continue; }
        lastLog = millis();

        extern bool mqttConnected();
        bool wifiUp = (WiFi.status() == WL_CONNECTED);
        bool mqttUp = mqttConnected();

        if (!wifiUp || !mqttUp) {
            // ── Buffer rows while offline ─────────────────────────
            char row[220];
            buildRow(row, sizeof(row));
            if (row[0] == '\0') continue;

            if (!_buffering || !_bufFile) {
                _bufFile = SD.open(BUFFER_FILE, FILE_APPEND);
                if (_bufFile) {
                    _buffering = true;
                    Serial.println("[SD] WiFi/MQTT down — buffering to SD card");
                } else {
                    Serial.println("[SD] Buffer open failed, skipping");
                    continue;
                }
            }
            if (!_bufFile.println(row)) {
                _bufFile.close();
                _buffering = false;
                sdRecover();
                continue;
            }
            if (++_bufFlush >= FLUSH_EVERY) { _bufFile.flush(); _bufFlush = 0; }
        } else {
            // ── WiFi+MQTT up — close buffer, mqttTask will replay ─
            if (_buffering) {
                if (_bufFile) { _bufFile.flush(); _bufFile.close(); }
                _buffering = false;
                Serial.println("[SD] WiFi+MQTT back — buffer closed, replay pending");
            }
        }
    }
}

        p["I"]  = phase[i].Irms;

        p["S"]  = phase[i].S;
        p["Q"]  = phase[i].Q;
        p["PF"] = phase[i].PF;    }

}

void wifiInit() {
    loadCredentials();
    if (strlen(_nvsSsid) == 0) {
        Serial.println("[WiFi] No saved credentials — skipping connect");
        return;
    }
    WiFi.mode(WIFI_STA);
    WiFi.begin(_nvsSsid, _nvsPass);
    Serial.printf("[WiFi] Connecting to %s", _nvsSsid);

    int tries = 0;
    while (WiFi.status() != WL_CONNECTED && tries < 20) {
        vTaskDelay(pdMS_TO_TICKS(500));
        Serial.print('.');
        tries++;
    }

    if (WiFi.status() == WL_CONNECTED) {
        _connected = true;
        Serial.printf("\n[WiFi] Connected — IP: %s\n",
                      WiFi.localIP().toString().c_str());

        // mDNS  →  http://calmed.local
        if (MDNS.begin(MDNS_NAME)) {
            MDNS.addService("http", "tcp", 80);
            Serial.printf("[mDNS] http://%s.local\n", MDNS_NAME);
        }

        // HTTP root → dashboard
        server.on("/", HTTP_GET, []() {
            server.sendHeader("Content-Encoding", "identity");
            server.send_P(200, "text/html", HTML);
        });

        // JSON snapshot endpoint
        server.on("/data", HTTP_GET, []() {
            JsonDocument doc;
            if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(10)) == pdTRUE) {
                doc["totalP"] = totalP; doc["totalS"] = totalS;
                doc["avgPF"]  = avgPF;  doc["vUnbal"] = vUnbal;
                JsonArray ph = doc["phases"].to<JsonArray>();
                for (int i = 0; i < 3; i++) {
                    JsonObject p = ph.add<JsonObject>();
                    p["V"] = phase[i].Vrms; p["I"] = phase[i].Irms;
                    p["P"] = phase[i].P;    p["PF"]= phase[i].PF;
                }
                xSemaphoreGive(dataMutex);
            }
            String out; serializeJson(doc, out);
            server.send(200, "application/json", out);
        });

        server.begin();
        ws.begin();
        ws.onEvent([](uint8_t, WStype_t, uint8_t*, size_t){});

    } else {
        Serial.println("\n[WiFi] Failed — running offline");
    }
}

void wifiTask(void* param) {
    vTaskDelay(pdMS_TO_TICKS(100));
    wifiInit();

    unsigned long lastBroadcast = 0;
    for (;;) {
        // Pause all WiFi management while screen_wifi is scanning or connecting
        if (wifiSwitching) {
            vTaskDelay(pdMS_TO_TICKS(100));
            continue;
        }

        if (_connected) {
            if (WiFi.status() != WL_CONNECTED) {
                _connected = false;
                Serial.println("[WiFi] Dropped — reconnecting...");
                ws.disconnect();
                WiFi.disconnect(false);
                vTaskDelay(pdMS_TO_TICKS(500));
                loadCredentials();   // re-read NVS — picks up FORGET / new network
                if (strlen(_nvsSsid) == 0) {
                    Serial.println("[WiFi] No credentials — waiting for user config");
                    vTaskDelay(pdMS_TO_TICKS(2000));
                    continue;
                }
                WiFi.begin(_nvsSsid, _nvsPass);
            } else {
                server.handleClient();
                ws.loop();
                if (millis() - lastBroadcast >= 1000) {
                    lastBroadcast = millis();
                    broadcastData();
                }
            }
        } else {
            if (WiFi.status() == WL_CONNECTED) {
                _connected = true;
                Serial.printf("[WiFi] Reconnected — %s\n",
                              WiFi.localIP().toString().c_str());
                // mDNS must restart; server + ws are already running — do NOT re-begin
                MDNS.end();
                if (MDNS.begin(MDNS_NAME)) {
                    MDNS.addService("http", "tcp", 80);
                }
            }
        }
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

    if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(5)) == pdTRUE) {

            _prevPF[ph] = snap[ph].PF;

static void drawSparkline(int x, int y, int w, int h,

    float mn = data[0], mx = data[0];
    for (int i = 1; i < len; i++) {
        mn = min(mn, data[i]);
        mx = max(mx, data[i]);
    }
    float range = mx - mn;

    tft.fillRect(x, y, w, h, C_DARK);

        int py = y + h - 2 - (int)(((data[(i + _pIdx) % len] - mn) / range) * (h - 4));

        prevPy = py;
    }
}

    if (millis() - _lastSpark >= 1000) {

            for (int p = 0; p < 3; p++)
                _pHist[p][_pIdx] = phase[p].P;

    }

    // Update display every 500ms
    if (millis() - _lastUpdate < 500) return;

    unsigned long sec = (millis() - energyStartMs) / 1000;
    tft.fillRect(0, 54, TFT_W, 14, C_BG);

             sec / 3600, (sec % 3600) / 60, sec % 60);
    tft.print(uptime);

    // Read energy snapshot
    float eWh[3], totWh;
    float pw[3];
    if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(5)) == pdTRUE) {
        for (int i = 0; i < 3; i++) {

            pw[i]  = phase[i].P;

        tft.setTextColor(C_YELLOW);
        tft.setCursor(158, ry + 10);

    tft.setTextColor(C_ACCENT); tft.setTextSize(1);

static int drawFaultLines(int startY, float vu, float apf, float f,
                           float maxVdev, float iMax, float rated) {
    int lineY = startY;

    if (vu >= lim_vunbal_crit) {

        snprintf(b, sizeof(b), "PF %.2f < %.2f limit", apf, lim_pf_crit);
        tft.setCursor(14, lineY); tft.print(b); lineY += 13;
    }
    if (f < FREQ_CRIT_LO || f > FREQ_CRIT_HI) {
        snprintf(b, sizeof(b), "Freq %.2f Hz (nom 50)", f);
        tft.setCursor(14, lineY); tft.print(b); lineY += 13;
    }
    if (maxVdev > lim_ov_pct) {
        snprintf(b, sizeof(b), "V-Dev %.1f%% > %.0f%% limit", maxVdev, lim_ov_pct);
        tft.setCursor(14, lineY); tft.print(b); lineY += 13;
    }
    if (iMax >= rated * OC_CRIT_FRAC) {
        snprintf(b, sizeof(b), "OC %.1fA >= %.0fA rated", iMax, rated);
        tft.setCursor(14, lineY); tft.print(b); lineY += 13;
    }
    return lineY;

void drawAlerts() {
    if (_init) { drawSkeleton(); _init = false; }

    PhaseData snap[3];
    float vu, iu, apf, f;
    if (xSemaphoreTake(dataMutex, pdMS_TO_TICKS(5)) == pdTRUE) {
        for (int i = 0; i < 3; i++) snap[i] = phase[i];
        vu = vUnbal; iu = iUnbal; apf = avgPF; f = freqHz;
        xSemaphoreGive(dataMutex);
    } else { return; }

    float nominal = (voltMode == 2) ? V_NOMINAL_400 : V_NOMINAL_230;
    float iRated  = (sensorMode >= 0 && sensorMode < 6)
                    ? SCT_RATED_A[sensorMode] : 30.0f;

    float iMax = 0;
    for (int i = 0; i < 3; i++) iMax = max(iMax, snap[i].Irms);

    float maxVdev = 0;
    for (int i = 0; i < 3; i++)
        maxVdev = max(maxVdev, fabsf(snap[i].Vrms - nominal) / nominal * 100.0f);

    bool anyFault = (vu  >= lim_vunbal_crit)              ||
                    (apf  < lim_pf_crit && apf > 0.01f)   ||
                    (f < FREQ_CRIT_LO || f > FREQ_CRIT_HI)||
                    (maxVdev > lim_ov_pct)                 ||
                    (iMax >= iRated * OC_CRIT_FRAC);

    if (millis() - _lastBlink >= 400) {
        _lastBlink  = millis();
        _blinkState = !_blinkState;
    }

    // ── Alert / OK box ────────────────────────────────────
    if (anyFault) {
        tft.fillRoundRect(8, 66, TFT_W-16, 90, 8,
                          _blinkState ? 0x4000 : C_DARK);
        tft.drawRoundRect(8, 66, TFT_W-16, 90, 8, C_RED);
        tft.setTextColor(_blinkState ? C_RED : C_ORANGE);
        tft.setTextSize(2);
        tft.setCursor(52, 74); tft.print("! ALERT !");
        tft.setTextColor(C_WHITE); tft.setTextSize(1);
        drawFaultLines(96, vu, apf, f, maxVdev, iMax, iRated);
    } else {
        tft.fillRoundRect(8, 66, TFT_W-16, 90, 8, C_DARK);
        tft.drawRoundRect(8, 66, TFT_W-16, 90, 8, C_GREEN);
        tft.setTextColor(C_GREEN); tft.setTextSize(2);
        tft.setCursor(36, 78); tft.print("  ALL OK  ");
        tft.setTextColor(C_GRAY); tft.setTextSize(1);
        tft.setCursor(14, 108); tft.print("No active faults detected");
        tft.setCursor(14, 122); tft.print("ESA monitoring running...");
        tft.setCursor(14, 136); tft.print("IEC 61000-4-30 compliant");
    }

    // ── Parameter rows ────────────────────────────────────
    int ry = 176;
    drawStatusRow(ry, "V-Unbalance",  vu,
                  lim_vunbal_warn, lim_vunbal_crit, false, "%");  ry += 14;
    drawStatusRow(ry, "Power Factor", apf,
                  lim_pf_warn, lim_pf_crit, true, "");            ry += 14;
    drawStatusRow(ry, "Frequency",    f,
                  FREQ_WARN_LO, FREQ_CRIT_LO, true, "Hz");        ry += 14;

    float maxOV = 0;
    for (int i = 0; i < 3; i++) {
        float dev = snap[i].Vrms - nominal;
        if (dev > maxOV) maxOV = dev;
    }
    float ovPct = maxOV / nominal * 100.0f;
    drawStatusRow(ry, "Overvoltage",  ovPct,
                  lim_ov_pct, lim_ov_crit, false, "%");    ry += 14;

    float maxUV = 0;
    for (int i = 0; i < 3; i++) {
        if (snap[i].Vrms < 5.0f) continue;
        float sag = nominal - snap[i].Vrms;
        if (sag > maxUV) maxUV = sag;
    }
    float uvPct = maxUV / nominal * 100.0f;
    drawStatusRow(ry, "Undervoltage", uvPct,
                  lim_uv_pct, lim_uv_crit, false, "%");    ry += 14;

    drawStatusRow(ry, "I-Unbalance",  iu,
                  lim_vunbal_warn, lim_vunbal_crit, false, "%");  ry += 14;

    drawOCRow(ry, snap[0].Irms, snap[1].Irms, snap[2].Irms, iRated);

    drawStatusBar();
}

            lim_vunbal_crit = lim_vunbal_warn + 0.5f;

            lim_pf_crit = lim_pf_warn - 0.01f;

            lim_pf_warn = lim_pf_crit + 0.01f;
        if (_selField == 6 && lim_iunbal_warn >= lim_iunbal_crit)
            lim_iunbal_crit = lim_iunbal_warn + 0.5f;
        if (_selField == 7 && lim_iunbal_crit <= lim_iunbal_warn)

    if (!_inputting) {

        int fy = 68;

                drawKeyboardOverlay();
                drawInputBox();
                drawKeyboardOverlay();

        }

    }

    int by = KB_Y_TOP + 16, bh = 18;
    int okx = TFT_W - 40;
    if (ty >= by && ty <= by + bh && tx >= okx) {
        commitInput();
        return;

                        _input[_inputLen++] = k[0];
                        _input[_inputLen]   = '\0';
                    }
                }
                drawInputBox();
                return;
            }
        }
    }
}

static void nvsLoad() {
    Preferences p;
    p.begin(NVS_NS, true);
    _savedCount = p.getInt("cnt", 0);
    if (_savedCount > MAX_SAVED) _savedCount = MAX_SAVED;
    for (int i = 0; i < _savedCount; i++) {
        char ks[5], kp[5];
        snprintf(ks, 5, "s%d", i);
        snprintf(kp, 5, "p%d", i);
        String s = p.getString(ks, "");
        String w = p.getString(kp, "");
        strncpy(_savedSSID[i], s.c_str(), 32); _savedSSID[i][32] = '\0';
        strncpy(_savedPass[i], w.c_str(), 64); _savedPass[i][64] = '\0';
    }
    p.end();
}

// Write all saved networks back to NVS (both new namespace and legacy)
static void nvsFlush() {
    // New namespace (multi-net)
    Preferences p;
    p.begin(NVS_NS, false);
    p.clear();
    p.putInt("cnt", _savedCount);
    for (int i = 0; i < _savedCount; i++) {
        char ks[5], kp[5];
        snprintf(ks, 5, "s%d", i);
        snprintf(kp, 5, "p%d", i);
        p.putString(ks, _savedSSID[i]);
        p.putString(kp, _savedPass[i]);
    }
    p.end();
    // Legacy namespace (so wifi_manager.cpp can still auto-connect on boot)
    // Always write the LAST connected network as the "primary"
    p.begin(NVS_OLD, false);
    if (_savedCount > 0) {
        p.putString("ssid", _savedSSID[_savedCount - 1]);
        p.putString("pass", _savedPass[_savedCount - 1]);
    } else {
        p.remove("ssid");
        p.remove("pass");
    }
    p.end();
}

// Add or update a network in the saved list
static void nvsSaveNetwork(const char* ssid, const char* pass) {
    // Check if already exists → update password
    for (int i = 0; i < _savedCount; i++) {
        if (strcmp(_savedSSID[i], ssid) == 0) {
            strncpy(_savedPass[i], pass, 64);
            _savedPass[i][64] = '\0';
            // Move to end (most recent)
            char ts[33], tp[65];
            strcpy(ts, _savedSSID[i]); strcpy(tp, _savedPass[i]);
            for (int j = i; j < _savedCount - 1; j++) {
                strcpy(_savedSSID[j], _savedSSID[j+1]);
                strcpy(_savedPass[j], _savedPass[j+1]);
            }
            strcpy(_savedSSID[_savedCount-1], ts);
            strcpy(_savedPass[_savedCount-1], tp);
            nvsFlush();
            return;
        }
    }
    // New network: push to end, evict oldest if full
    if (_savedCount < MAX_SAVED) {
        strncpy(_savedSSID[_savedCount], ssid, 32); _savedSSID[_savedCount][32] = '\0';
        strncpy(_savedPass[_savedCount], pass, 64); _savedPass[_savedCount][64] = '\0';
        _savedCount++;
    } else {
        // Shift left (drop oldest) and add at end
        for (int i = 0; i < MAX_SAVED - 1; i++) {
            strcpy(_savedSSID[i], _savedSSID[i+1]);
            strcpy(_savedPass[i], _savedPass[i+1]);
        }
        strncpy(_savedSSID[MAX_SAVED-1], ssid, 32); _savedSSID[MAX_SAVED-1][32] = '\0';
        strncpy(_savedPass[MAX_SAVED-1], pass, 64); _savedPass[MAX_SAVED-1][64] = '\0';
    }
    nvsFlush();
}

// Remove a saved network by index
static void nvsForget(int idx) {
    if (idx < 0 || idx >= _savedCount) return;
    for (int i = idx; i < _savedCount - 1; i++) {
        strcpy(_savedSSID[i], _savedSSID[i+1]);
        strcpy(_savedPass[i], _savedPass[i+1]);
    }
    _savedCount--;
    nvsFlush();
}

// Returns index in _savedSSID[] if ssid is saved, -1 otherwise
static int savedIdx(const char* ssid) {
    for (int i = 0; i < _savedCount; i++)
        if (strcmp(_savedSSID[i], ssid) == 0) return i;
    return -1;

// ════════════════════════════════════════════════════════════
static void calcLayout() {
    // Saved section height
    int savedH = (_savedCount > 0) ? (14 + _savedCount * SAVED_ROW) : 0;
    // Where scan list starts (after header + saved section + "AVAILABLE" label)
    _listY   = HDR_BOT + savedH + 14;
    // How many scan rows fit before the back button (TFT_H - 30)
    int avail = (TFT_H - 30) - _listY;
    _visible = avail / SCAN_ROW;
    if (_visible < 2) _visible = 2;
    if (_visible > 6) _visible = 6;
}

// ════════════════════════════════════════════════════════════
//  DRAW HELPERS
// ════════════════════════════════════════════════════════════
static void drawSignalBars(int x, int y, int32_t rssi, uint16_t col) {
    int bars = 1;
    if      (rssi >= -55) bars = 4;
    else if (rssi >= -65) bars = 3;
    else if (rssi >= -75) bars = 2;
    for (int b = 0; b < 4; b++) {
        int bh = 4 + b * 3;
        tft.fillRect(x + b * 5, y - bh, 3, bh, b < bars ? col : C_DARK);
    }
}

static void drawLock(int x, int y, uint16_t col) {
    tft.drawRoundRect(x, y + 4, 8, 7, 2, col);
    tft.drawCircle(x + 4, y + 4, 3, col);
    tft.fillRect(x + 1, y + 2, 6, 4, C_BG);
}

// Small label badge  e.g. "CONNECTED" or "SAVED"
static void drawBadge(int x, int y, const char* text, uint16_t fg, uint16_t bg) {
    int w = strlen(text) * 6 + 6;
    tft.fillRoundRect(x, y, w, 12, 3, bg);
    tft.setTextColor(fg); tft.setTextSize(1);
    tft.setCursor(x + 3, y + 2); tft.print(text);
}

                               strcmp(_savedSSID[i], WiFi.SSID().c_str()) == 0);

                tft.setTextColor(isConn ? C_GREEN : C_WHITE); tft.setTextSize(1);
                tft.setCursor(10, ry + 5); tft.print(buf);

                } else {
                    drawBadge(10, ry + 17, "SAVED", C_GRAY, C_BORDER);
                }

                tft.fillRoundRect(TFT_W - 54, ry + 7, 48, 16, 4, C_DARK);
                tft.drawRoundRect(TFT_W - 54, ry + 7, 48, 16, 4, C_RED);
                tft.setTextColor(C_RED); tft.setCursor(TFT_W - 46, ry + 11);
                tft.print("FORGET");
            }
        }

        // ── AVAILABLE SECTION LABEL ────────────────────────
        int ay = HDR_BOT + (_savedCount > 0 ? 14 + _savedCount * SAVED_ROW : 0);
        tft.fillRect(0, ay, TFT_W, 14, C_DARK);
        tft.drawFastHLine(0, ay, TFT_W, C_BORDER);
        tft.setTextColor(C_GRAY); tft.setTextSize(1);
        tft.setCursor(4, ay + 3);
        tft.print(_scanning ? "Scanning..." : (_netCount == 0 ? "AVAILABLE  (tap SCAN)" : "AVAILABLE"));
    }  // end full

    if (_scanning) {
        tft.fillRect(0, _listY, TFT_W, TFT_H - _listY - 30, C_BG);
        static int dot = 0; dot = (dot + 1) % 4;
        for (int d = 0; d < 4; d++)
            tft.fillCircle(80 + d * 14, _listY + 40, 3, d <= dot ? C_ACCENT : C_DARK);

        tft.setTextColor(C_GRAY); tft.setTextSize(1);
        tft.setCursor(30, _listY + 20); tft.print("No networks found. Tap SCAN.");
        return;
    }

    int shown = min(_netCount, _visible);
    for (int i = 0; i < shown; i++) {
        int ni = i + _scroll;
        if (ni >= _netCount) break;
        int ry  = _listY + i * SCAN_ROW;
        bool sel = (ni == _selected);
        int si  = savedIdx(_ssid[ni]);   // -1 if not saved

        uint16_t rowBg = sel ? C_PANEL : C_BG;
        tft.fillRect(0, ry, TFT_W, SCAN_ROW - 1, rowBg);
        tft.drawFastHLine(0, ry + SCAN_ROW - 1, TFT_W, C_BORDER);

        uint16_t ic = sel ? C_ACCENT : C_GRAY;

        if (sel) tft.fillRect(0, ry, 3, SCAN_ROW - 1, C_ACCENT);

        // Signal bars
        drawSignalBars(8, ry + SCAN_ROW - 6, _rssi[ni], ic);

        // Lock icon
        if (_secured[ni]) drawLock(30, ry + 5, ic);

        // SSID
        char buf[22]; strncpy(buf, _ssid[ni], 21); buf[21] = '\0';
        tft.setTextColor(tc); tft.setTextSize(1);
        tft.setCursor(46, ry + 6); tft.print(buf);

        // Sub-text
        char sub[28];
        snprintf(sub, sizeof(sub), "%d dBm  %s", _rssi[ni], _secured[ni] ? "Secured" : "Open");
        tft.setTextColor(C_GRAY); tft.setCursor(46, ry + 18); tft.print(sub);

        // Saved checkmark
        if (si >= 0) {
            tft.setTextColor(C_GREEN); tft.setCursor(TFT_W - 16, ry + 12);
            tft.print("\x03");  // custom glyph → just use a dot
            tft.fillCircle(TFT_W - 10, ry + 12, 4, C_GREEN);
        } else {
            tft.setTextColor(C_BORDER); tft.setCursor(TFT_W - 14, ry + 11);
            tft.print(">");
        }
    }

    int below = _listY + shown * SCAN_ROW;
    if (below < TFT_H - 30) tft.fillRect(0, below, TFT_W, TFT_H - 30 - below, C_BG);

    // Scroll arrows
    if (_scroll > 0) {
        tft.setTextColor(C_ACCENT); tft.setCursor(114, _listY - 8); tft.print("^");
    }
    if (_scroll + _visible < _netCount) {
        tft.setTextColor(C_ACCENT); tft.setCursor(114, below + 2); tft.print("v");
    }
}

static bool handleKeyTap(int tx, int ty) {
    if (tx >= TFT_W - 46 && tx <= TFT_W - 6 && ty >= 50 && ty <= 76) {
        if (_passLen > 0) _pass[--_passLen] = '\0';
        return true;
    }
    int ky4 = KB_Y0 + 3 * KB_ROW_H + 2;
    if (ty >= ky4 && ty <= ky4 + KB_ROW_H) {
        if (tx < 48)               { _state = STATE_LIST; _init = true; return true; }
        if (tx >= 50 && tx <= 150) { if (_passLen < 64) { _pass[_passLen++] = ' '; _pass[_passLen] = '\0'; } return true; }
        if (tx >= 154)             { _state = STATE_CONNECTING; return true; }
    }
    {
        const char* r = _numbers ? KB_N3 : (_shifted ? KB_HI[2] : KB_LO[2]);
        int len = strlen(r), kw = (TFT_W - 4) / (len + 2), ky = KB_Y0 + 2 * KB_ROW_H;
        if (ty >= ky && ty < ky + KB_ROW_H) {
            if (tx < kw * 2 + 2) {
                if (!_numbers) { if (!_shifted) _shifted = true; else { _shifted = false; _numbers = true; } }
                else           { _numbers = false; _shifted = false; }
                return true;
            }
            for (int k = 0; k < len; k++) {
                int kx = 2 + (k + 2) * kw;
                if (tx >= kx && tx < kx + kw - 2) {
                    if (_passLen < 64) { _pass[_passLen++] = r[k]; _pass[_passLen] = '\0'; if (_shifted) _shifted = false; }
                    return true;
                }
            }
            // bottom delete removed — right edge of row 3 is dead zone
        }
    }
    for (int row = 0; row < 2; row++) {
        const char* r = _numbers ? (row == 0 ? KB_N1 : KB_N2)
                                 : (row == 0 ? (_shifted ? KB_HI[0] : KB_LO[0])
                                             : (_shifted ? KB_HI[1] : KB_LO[1]));
        int len = strlen(r), kw = (TFT_W - 4) / len, ky = KB_Y0 + row * KB_ROW_H;
        if (ty >= ky && ty < ky + KB_ROW_H) {
            int k = (tx - 2) / kw;
            if (k >= 0 && k < len && _passLen < 64) {
                _pass[_passLen++] = r[k]; _pass[_passLen] = '\0';
                if (_shifted) _shifted = false;
            }
            return true;
        }
    }
    return false;
}

static void doConnect() {
    wifiSwitching = true;

    // Clean disconnect — do NOT erase credentials (false)
    WiFi.disconnect(false);
    vTaskDelay(pdMS_TO_TICKS(400));
    WiFi.mode(WIFI_STA);
    vTaskDelay(pdMS_TO_TICKS(100));

    WiFi.begin(_ssid[_selected], _pass);

    int tries = 0;
    while (WiFi.status() != WL_CONNECTED && tries < 20) {
        delay(500);
        drawConnecting();
        tries++;
    }

    if (WiFi.status() == WL_CONNECTED) {
        _connOK = true;
        strncpy(_resultIP, WiFi.localIP().toString().c_str(), 19);
        nvsSaveNetwork(_ssid[_selected], _pass);   // save to NVS + update memory list
        nvsLoad();                                  // reload _savedSSID array to reflect new entry
        MDNS.end();
        MDNS.begin("calmed");
    } else {
        _connOK = false;
        WiFi.disconnect(false);
    }

    wifiSwitching = false;
    _state = STATE_RESULT;
    _init  = true;
}

// ════════════════════════════════════════════════════════════
//  SCAN
// ════════════════════════════════════════════════════════════
static void doScan() {
    _scanning = true; _netCount = 0; _selected = -1; _scroll = 0;
    drawList(true);

    wifiSwitching = true; 

    // Always async so we can animate and scanComplete() works for both connected/offline
    WiFi.mode(WIFI_STA);
    WiFi.disconnect(true);
    delay(150); 
    WiFi.scanDelete();                              // clear any stale previous scan
    WiFi.scanNetworks(true /*async*/);

    unsigned long t0 = millis();
    while (WiFi.scanComplete() < 0 && millis() - t0 < 8000) {
        drawList(false);
        delay(300);
    }

    int n = WiFi.scanComplete();
    if (n > 0) {
        struct Net { char ssid[33]; int32_t rssi; bool sec; };
        Net tmp[MAX_NETS * 2]; int total = min(n, MAX_NETS * 2);
        for (int i = 0; i < total; i++) {
            strncpy(tmp[i].ssid, WiFi.SSID(i).c_str(), 32); tmp[i].ssid[32] = '\0';
            tmp[i].rssi = WiFi.RSSI(i);
            tmp[i].sec  = (WiFi.encryptionType(i) != WIFI_AUTH_OPEN);
        }
        // Sort by RSSI desc
        for (int i = 0; i < total - 1; i++)
            for (int j = 0; j < total - 1 - i; j++)
                if (tmp[j].rssi < tmp[j+1].rssi) { Net t = tmp[j]; tmp[j] = tmp[j+1]; tmp[j+1] = t; }
        // Deduplicate + fill
        _netCount = 0;
        for (int i = 0; i < total && _netCount < MAX_NETS; i++) {
            bool dup = false;
            for (int j = 0; j < _netCount; j++)
                if (strcmp(_ssid[j], tmp[i].ssid) == 0) { dup = true; break; }
            if (!dup) {
                strncpy(_ssid[_netCount], tmp[i].ssid, 32); _ssid[_netCount][32] = '\0';
                _rssi[_netCount] = tmp[i].rssi; _secured[_netCount] = tmp[i].sec;
                _netCount++;
            }
        }
    }
    WiFi.scanDelete();
    _scanning = false;
    wifiSwitching = false;  
    drawList(true);
}

// ════════════════════════════════════════════════════════════
//  PUBLIC API
// ════════════════════════════════════════════════════════════
void resetWifi() {
    _state = STATE_LIST; _init = true; _scanning = false;
    _netCount = 0; _selected = -1; _scroll = 0;
    _passLen = 0; _pass[0] = '\0'; _shifted = false; _numbers = false;
    nvsLoad();   // always reload saved networks on screen open
}

void drawWifi() {
    switch (_state) {
        case STATE_LIST:
            if (_scanning) { drawList(false); return; }
            if (_init) { drawList(true); _init = false; }
            break;
        case STATE_KB:
            if (_init) { drawKeyboard(true); _init = false; }
            else        drawKeyboard(false);
            break;
        case STATE_CONNECTING:
            drawConnecting();
            doConnect();
            break;
        case STATE_RESULT:
            if (_init) { drawResult(); _init = false; }
            break;
    }
}

void handleWifi(int tx, int ty) {
    switch (_state) {

        case STATE_LIST: {
            // ── Back button ──────────────────────────────────────
            if (backTapped(tx, ty)) {
                resetWifi(); resetSettings(); gotoScreen(SCR_SETTINGS); return;
            }
            // ── SCAN button ──────────────────────────────────────
            if (tx >= TFT_W - 56 && ty >= HDR_BOT - 20 && ty <= HDR_BOT - 4) {
                ripple(tx, ty, C_ACCENT); doScan(); return;
            }
            calcLayout();

            // ── FORGET buttons (saved section) ───────────────────
            if (_savedCount > 0) {
                int sy = HDR_BOT + 14;
                for (int i = 0; i < _savedCount; i++) {
                    int ry = sy + i * SAVED_ROW;
                    // FORGET button: x in [TFT_W-54, TFT_W-6], y in [ry+7, ry+23]
                    if (tx >= TFT_W - 54 && ty >= ry + 7 && ty <= ry + 23) {
                        ripple(tx, ty, C_RED);
                        // If forgetting the currently connected network, disconnect it
                        if (WiFi.status() == WL_CONNECTED &&
                            strcmp(_savedSSID[i], WiFi.SSID().c_str()) == 0) {
                            WiFi.disconnect(true);   // true = also erase stored credentials
                        }
                        nvsForget(i);
                        nvsLoad();          // refresh memory from NVS
                        _init = true;
                        drawList(true);
                        return;
                    }
                }
            }

            // ── Tap on scan result row ───────────────────────────
            if (ty >= _listY && _netCount > 0 && !_scanning) {
                int ni = ((ty - _listY) / SCAN_ROW) + _scroll;
                if (ni >= 0 && ni < _netCount) {
                    _selected = ni;
                    int si = savedIdx(_ssid[ni]);
                    if (si >= 0) {
                        // Already saved → connect directly without password keyboard
                        strncpy(_pass, _savedPass[si], 64);
                        _passLen = strlen(_pass);
                        _state = STATE_CONNECTING;
                    } else if (_secured[ni]) {
                        // New secured network → ask for password
                        _passLen = 0; _pass[0] = '\0';
                        _shifted = false; _numbers = false;
                        _state = STATE_KB; _init = true;
                    } else {
                        // Open network → connect directly
                        _pass[0] = '\0'; _passLen = 0;
                        _state = STATE_CONNECTING;
                    }
                    drawList(false);
                }
            }
            break;
        }

        case STATE_KB:
            if (handleKeyTap(tx, ty)) {
                if (_state == STATE_KB) { _init = true; drawKeyboard(true); }
            }
            break;

        case STATE_RESULT:
            if (_connOK) {
                if (btnTapped(tx, ty, 60, 240, 120, 30)) {
                    resetWifi(); resetSettings(); gotoScreen(SCR_SETTINGS);
                }
            } else {
                if (btnTapped(tx, ty, 20, 240, 90, 30)) {
                    // RETRY → back to keyboard
                    _passLen = 0; _pass[0] = '\0';
                    _state = STATE_KB; _init = true;
                }
                if (btnTapped(tx, ty, 130, 240, 90, 30)) {
                    // BACK → list
                    _state = STATE_LIST; _init = true;
                }
            }
            break;

        default: break;
    }
}
